cmake_minimum_required(VERSION 3.13)
project(pcss_contract_tests LANGUAGES CXX)
# Standalone test target: does not require the full slicer dependency stack.
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
set(PCSS_GUI_DIR "${CMAKE_CURRENT_LIST_DIR}/../../src/slic3r/GUI")
add_executable(pcss_math_tests test_math.cpp)
target_include_directories(pcss_math_tests PRIVATE "${PCSS_GUI_DIR}")
add_test(NAME pcss_math COMMAND pcss_math_tests)
option(PCSS_TEST_OPENGL "Compile the actual renderer and execute GPU readback tests" OFF)
if(PCSS_TEST_OPENGL)
    find_package(OpenGL REQUIRED)
    find_package(GLEW REQUIRED)
    find_package(glfw3 REQUIRED)
    add_executable(pcss_gl_tests test_gl.cpp "${PCSS_GUI_DIR}/PCSSShadowRenderer.cpp")
    target_include_directories(pcss_gl_tests PRIVATE "${PCSS_GUI_DIR}")
    target_link_libraries(pcss_gl_tests PRIVATE GLEW::GLEW glfw OpenGL::GL)
    target_compile_definitions(pcss_gl_tests PRIVATE PCSS_SHADER_DIRECTORY="${CMAKE_CURRENT_LIST_DIR}/../../resources/shaders/140")
    add_test(NAME pcss_gl COMMAND pcss_gl_tests)
endif()
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
    add_test(NAME pcss_source_contracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/test_integration.py")
endif()
